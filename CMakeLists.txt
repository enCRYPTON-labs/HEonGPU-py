cmake_minimum_required(VERSION 4.0 FATAL_ERROR)

if(NOT DEFINED SKBUILD_PROJECT_VERSION)
    set(SKBUILD_PROJECT_VERSION 0.1.0)
endif()

project(heongpu_python VERSION ${SKBUILD_PROJECT_VERSION} LANGUAGES C CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Choose the type of build." FORCE)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(HEONGPU_PY_HEONGPU_PROVIDER "AUTO" CACHE STRING
    "Where HEonGPU comes from: AUTO (an installed HEonGPU, otherwise built from source), SYSTEM (an installed HEonGPU only) or FETCH (always built from source)")
set_property(CACHE HEONGPU_PY_HEONGPU_PROVIDER PROPERTY STRINGS AUTO SYSTEM FETCH)
set(HEONGPU_PY_HEONGPU_REPOSITORY "https://github.com/enCRYPTON-labs/HEonGPU.git"
    CACHE STRING "Git repository of HEonGPU, used when HEonGPU is built from source")
set(HEONGPU_PY_HEONGPU_TAG "v2.1.1"
    CACHE STRING "Git tag of HEonGPU, used when HEonGPU is built from source")
option(HEONGPU_PY_STUBS "Generate the type stubs (.pyi) of the extension module" ON)

set(HEONGPU_PY_COMPONENT heongpu_python)

include(cmake/HEonGPUDependency.cmake)

# Python and nanobind.
# SKBUILD_SABI_COMPONENT is Development.SABIModule when scikit-build-core
# builds a wheel of the stable ABI.
find_package(Python 3.10 REQUIRED COMPONENTS Interpreter Development.Module
    ${SKBUILD_SABI_COMPONENT})
execute_process(
    COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir
    OUTPUT_STRIP_TRAILING_WHITESPACE
    OUTPUT_VARIABLE nanobind_ROOT
    COMMAND_ERROR_IS_FATAL ANY)
find_package(nanobind CONFIG REQUIRED)

file(GLOB HEONGPU_PY_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/cpp/*.cpp")

# STABLE_ABI builds a module of the stable ABI (abi3) for Python 3.12 and newer,
# and has no effect on older versions.
nanobind_add_module(_heongpu NB_STATIC STABLE_ABI ${HEONGPU_PY_SOURCES})
target_include_directories(_heongpu PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/cpp")
# HEonGPU links libraries the module does not use, such as cuRAND (more than
# 100 MB), which would otherwise be copied into the wheels.
target_link_options(_heongpu PRIVATE "LINKER:--as-needed")
target_link_libraries(_heongpu PRIVATE ${HEONGPU_PY_HEONGPU_TARGET} CUDA::cudart)
target_compile_definitions(_heongpu PRIVATE
    HEONGPU_PY_VERSION="${PROJECT_VERSION}"
    HEONGPU_PY_HEONGPU_VERSION="${HEONGPU_PY_HEONGPU_VERSION}")
heongpu_py_configure_device_link(_heongpu)

install(TARGETS _heongpu
    LIBRARY DESTINATION heongpu
    COMPONENT ${HEONGPU_PY_COMPONENT})

if(HEONGPU_PY_STUBS)
    # The stubs are generated by importing the module; it only touches the GPU
    # when it is used, so this works on machines without one.
    nanobind_add_stub(_heongpu_stub
        MODULE heongpu._heongpu
        RECURSIVE
        OUTPUT_PATH "${CMAKE_CURRENT_BINARY_DIR}/stubs/heongpu"
        PYTHON_PATH "${CMAKE_CURRENT_BINARY_DIR}/stub_path"
        MARKER_FILE "${CMAKE_CURRENT_BINARY_DIR}/stubs/heongpu/py.typed"
        DEPENDS _heongpu)
    # The module is imported as heongpu._heongpu without the Python sources of
    # the package, which would import it themselves.
    add_custom_command(TARGET _heongpu POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory
            "${CMAKE_CURRENT_BINARY_DIR}/stub_path/heongpu"
        COMMAND ${CMAKE_COMMAND} -E touch
            "${CMAKE_CURRENT_BINARY_DIR}/stub_path/heongpu/__init__.py"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:_heongpu>"
            "${CMAKE_CURRENT_BINARY_DIR}/stub_path/heongpu/"
        VERBATIM)
    install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/stubs/heongpu/"
        DESTINATION heongpu
        COMPONENT ${HEONGPU_PY_COMPONENT}
        FILES_MATCHING PATTERN "*.pyi")
endif()

heongpu_py_install_runtime_libraries()
